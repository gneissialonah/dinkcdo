// DinkCDO Core Booking Logic
// Settings: Operating hours = 12AM‑11PM; show only available slots; 3% commission (added on top)
const SETTINGS = {
  startHour: 0, // 12 AM (00:00)
  endHour: 23,  // 11 PM (23:00)
  commissionRate: 0.03 // UPDATED: 3% platform service fee
};

// Sample venue data (dynamic per owner‑registered court count)
const venues = [
  {
    id: "venue‑1",
    name: "Pickle World Limketkai",
    totalCourts: 3,
    baseFeePerHour: 500,
    bookings: {} // stored time‑slot bookings
  }
];

// Generate hourly time slots (0‑23)
function generateTimeSlots() {
  let slots = [];
  for(let h = SETTINGS.startHour; h <= SETTINGS.endHour; h++) {
    slots.push(h);
  }
  return slots;
}

// Calculate available courts per time slot → return only slots with >0 available
function getAvailableSlots(venue) {
  const allSlots = generateTimeSlots();
  const availableList = [];

  allSlots.forEach(hour => {
    const booked = venue.bookings[hour] || 0;
    const available = venue.totalCourts - booked;
    if(available > 0) {
      availableList.push({
        hour: hour,
        availableCourts: available,
        baseFee: venue.baseFeePerHour,
        totalPrice: (venue.baseFeePerHour * (1 + SETTINGS.commissionRate)).toFixed(2)
      });
    }
  });
  return availableList;
}

// Render venue list + available slots (only show available times)
function renderCourts() {
  const container = document.getElementById("court‑list");
  if(!container) return;
  container.innerHTML = "";

  venues.forEach(venue => {
    const venueDiv = document.createElement("div");
    venueDiv.className = "venue‑card";
    const slots = getAvailableSlots(venue);

    let slotHtml = "";
    slots.forEach(s => {
      const timeLabel = formatHour(s.hour);
      slotHtml += `<div class="time‑slot">${timeLabel} | ${s.availableCourts} court(s) available | ₱${s.totalPrice}</div>`;
    });

    venueDiv.innerHTML = `
      <h3>${venue.name}</h3>
      <p>Total courts: ${venue.totalCourts}</p>
      <div class="slot‑list">${slotHtml || "<p>No available slots today</p>"}</div>
      <a href="booking‑page.html?venue=${venue.id}" class="btn‑book">Book now</a>
    `;
    container.appendChild(venueDiv);
  });
}

function formatHour(h) {
  if(h === 0) return "12:00 AM‑1:00 AM";
  if(h < 11) return `${h}:00 AM‑${h+1}:00 AM`;
  if(h === 11) return "11:00 AM‑12:00 PM";
  if(h === 12) return "12:00 PM‑1:00 PM";
  if(h < 23) return `${h‑12}:00 PM‑${h‑11}:00 PM`;
  return "11:00 PM‑12:00 AM";
}

// Initialize on load
window.addEventListener("DOMContentLoaded", renderCourts);

// DinkCDO Player‑Owner Chat Logic (Prototype)
let chatWindow = document.getElementById("chat‑window");

// Send player message
function sendMessage() {
  const input = document.getElementById("chat‑message");
  const text = input.value.trim();
  if (!text) return;

  // Add player bubble
  chatWindow.innerHTML += `
    <div class="msg‑player">
      <div class="bubble‑player">${escapeHtml(text)}</div>
    </div>
  `;
  input.value = "";
  scrollChatToBottom();

  // Simulate owner auto‑reply (for demo only)
  setTimeout(() => {
    chatWindow.innerHTML += `
      <div class="msg‑owner">
        <div class="bubble‑owner">Thanks for your message — we will get back to you shortly.</div>
      </div>
    `;
    scrollChatToBottom();
  }, 1200);
}

function scrollChatToBottom() {
  chatWindow.scrollTop = chatWindow.scrollHeight;
}

// Prevent HTML injection (basic escape)
function escapeHtml(str) {
  return str.replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[c]));
}
